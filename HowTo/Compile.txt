To compile the LRIM simulation: 

# First time only — create the build directory and configure
cmake -S . -B build -DCMAKE_BUILD_TYPE=Release

# Build (repeat this after any code change)
cmake --build build -j$(nproc)

# Run
./build/run --solver metropolis --L 16 --sigma 1.5 --sigma_index 0

# Clean reconfigure after restructuring: 
rm -rf build
cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build -j$(nproc)
